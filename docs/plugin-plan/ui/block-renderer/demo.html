1宿主默认 · 普通代码块 / 未知语言
给我一个带退避的重试函数。

核心是指数退避 + 保留最后一次的错误：

ts 
export async function retry<T>(
  run: () => Promise<T>,
  times = 3,
): Promise<T> {
  let lastError: unknown;
  for (let attempt = 0; attempt < times; attempt++) {
    try {
      return await run();
    } catch (error) {
      lastError = error;
      await sleep(2 ** attempt * 250);
    }
  }
  throw lastError;
}
